업무처리 가이드 검색

function ask(){ const q = document.getElementById("q").value; const result = document.getElementById("result"); if(!q) return; const text = GUIDE.toLowerCase(); const keyword = q.toLowerCase(); if(text.includes(keyword)){ result.innerText = "관련 내용 발견:\n\n" + GUIDE.split("\n").filter(v=>v.toLowerCase().includes(keyword)).join("\n"); }else{ result.innerText = "관련 내용 없음"; } }
